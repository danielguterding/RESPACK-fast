cmake_minimum_required(VERSION 2.8)
if(${CMAKE_PROJECT_NAME} STREQUAL "Project")
  message(FATAL_ERROR "cmake should be executed not for 'src' subdirectory, but for the top directory of RESPACK.")
endif(${CMAKE_PROJECT_NAME} STREQUAL "Project")

set(SOURCES_COMMON inv.F90 est_NTK.f90 est_nwx2.f90 calc_int_sub.F m_rdinput.f90 m_fft3d_20150826.f90 int_fast.f90)
# fork: FFTW3 for m_int_fast (same lookup as src/chiqw)
set(FFTW3_ROOT "" CACHE PATH "FFTW3 installation prefix (include/fftw3.f03, lib/libfftw3)")
find_path(FFTW3_INCLUDE_DIR fftw3.f03 HINTS ${FFTW3_ROOT}/include)
find_library(FFTW3_LIBRARY NAMES fftw3 HINTS ${FFTW3_ROOT}/lib NO_DEFAULT_PATH)
if(NOT FFTW3_LIBRARY)
  find_library(FFTW3_LIBRARY NAMES fftw3)
endif()
if(NOT FFTW3_INCLUDE_DIR OR NOT FFTW3_LIBRARY)
  message(FATAL_ERROR "FFTW3 not found: set -DFFTW3_ROOT=<prefix>")
endif()
include_directories(${FFTW3_INCLUDE_DIR})
set_source_files_properties(int_fast.f90 PROPERTIES COMPILE_FLAGS "-fno-lto")
add_custom_target(int ALL)
add_library(intcom STATIC ${SOURCES_COMMON})
target_link_libraries(intcom ${FFTW3_LIBRARY} ${LAPACK_LIBRARIES} m)
add_executable(calc_w3d calc_intW.F wrt_model_w3d.f90)
add_dependencies(int calc_w3d)
target_link_libraries(calc_w3d  intcom )
add_executable(calc_j3d calc_intJ.F wrt_model_j3d.f90)
target_link_libraries(calc_j3d intcom )
add_dependencies(int calc_j3d)

install(TARGETS calc_w3d RUNTIME DESTINATION bin)
install(TARGETS calc_j3d RUNTIME DESTINATION bin)
